use the component registry /src/react/lib files to add dynamic react component

import type { ComponentConfig, ComponentRegistry } from '@/types/components'

class ComponentLibrary {
  private registry: ComponentRegistry = new Map()

  register(name: string, config: ComponentConfig) {
    console.log(`Registering component: ${name}`)
    this.registry.set(name, config)
    return this
  }

  get(name: string) {
    const component = this.registry.get(name)
    if (!component) {
      console.error(`Component not found: ${name}`)
      throw new Error(`Component "${name}" not found in registry`)
    }
    console.log(`Component loaded: ${name}`)
    return component
  }

  has(name: string): boolean {
    return this.registry.has(name)
  }

  list(): string[] {
    return Array.from(this.registry.keys())
  }

  getMetadata(name: string) {
    return this.get(name).metadata || {}
  }
}

export const componentLib = new ComponentLibrary()

Guide: Dynamic React Components in Astro with AppClient Wrapper
Based on the project analysis, here's how to implement dynamic React component loading with the AppClient wrapper in your Astro + React setup:

1. Dynamic Route Setup
Create a new dynamic route at src/pages/[...path].astro to handle all dynamic component loading:

astro
---
// src/pages/[...path].astro
import Layout from '@/layouts/ThemeLayout.astro';
import AppClient from '@/react/AppClient';
import { componentLib } from '@/react/lib/componentLib';
import type { AppClientProps } from '@/types/components';

const { path } = Astro.params;
const componentName = path?.join('/') || 'Home';

// Check if component exists
const componentExists = componentLib.has(componentName);
const metadata = componentExists ? componentLib.getMetadata(componentName) : null;

// Prepare client props
const clientProps: AppClientProps = {
  componentName,
  showThemeToggle: true,
  showToaster: true
};
---

<Layout title={metadata?.title || 'Dynamic Component'}>
  {componentExists ? (
    <AppClient client:load {...clientProps} />
  ) : (
    <div class="text-center py-12">
      <p class="text-destructive text-lg">
        The requested component "{componentName}" was not found.
      </p>
    </div>
  )}
</Layout>
2. Component Registration
Register your React components in the component library:

typescript
// src/react/lib/componentLib.ts
import type { ComponentType } from 'react';

interface ComponentMetadata {
  title: string;
  description?: string;
  category?: string;
}

const componentRegistry = new Map<string, {
  component: ComponentType<any>;
  metadata?: ComponentMetadata;
}>();

// Example registration
export function registerComponent(
  name: string, 
  component: ComponentType<any>, 
  metadata?: ComponentMetadata
) {
  componentRegistry.set(name, { component, metadata });
}

// Check if component exists
export function hasComponent(name: string): boolean {
  return componentRegistry.has(name);
}

// Get component
export function getComponent(name: string) {
  return componentRegistry.get(name)?.component;
}

// Get component metadata
export function getMetadata(name: string) {
  return componentRegistry.get(name)?.metadata;
}

// Register all components
import Button from '@/react/components/Button';
import Card from '@/react/components/Card';
// Import other components...

// Register components
registerComponent('Button', Button, {
  title: 'Button',
  description: 'A customizable button component',
  category: 'Basic'
});

registerComponent('Card', Card, {
  title: 'Card',
  description: 'A card component',
  category: 'Layout'
});

// Export the component library
export const componentLib = {
  has: hasComponent,
  get: getComponent,
  getMetadata,
  register: registerComponent
};
3. Using AppClient with Dynamic Components
The 
AppClient
 component is already set up to handle dynamic components. Here's how it works:

It accepts a componentName prop that matches a registered component
It provides essential providers (Theme, Query, etc.)
It handles loading and error states
4. Creating a New Dynamic Page
To create a new page that uses a dynamic component:

Register your component in componentLib.ts
Access it via the dynamic route: /your-component-name
5. Direct Component Usage
For direct usage in an Astro file:

astro
---
// src/pages/index.astro
import AppClient from '@/react/AppClient';
---

<AppClient 
  client:load 
  componentName="YourComponent" 
  showThemeToggle={true}
/>
6. Best Practices
Lazy Loading: Components are automatically lazy-loaded by the 
AppClient
Error Boundaries: Consider adding error boundaries in 
AppClient
Type Safety: Ensure all components have proper TypeScript types
Performance: Use client:visible or client:idle for below-the-fold components
Testing: Test your dynamic components in isolation
7. Advanced: Custom Providers
To add custom providers:

typescript
// In your page or layout
import { AuthProvider } from '@/react/providers/authContext';

// In your client props
const clientProps: AppClientProps = {
  componentName: 'YourComponent',
  additionalProviders: [AuthProvider]
};
This setup provides a robust foundation for dynamic component loading in your Astro + React application while maintaining type safety and developer experience.
